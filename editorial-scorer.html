    let activeTab = 'url';
    function switchTab(tab) {
    activeTab = tab;
    document.querySelectorAll('.tab-btn').forEach((b, i) => b.classList.toggle('active', (i === 0 && tab === 'url') || (i === 1 && tab === 'paste')));
    document.getElementById('tab-url').classList.toggle('active', tab === 'url');
    document.getElementById('tab-paste').classList.toggle('active', tab === 'paste');
    }
    const SYSTEM = (audience) => `You are an editorial quality analyst. Score the page content against these editorial principles, calibrated for the stated intended audience.
    INTENDED AUDIENCE: "${audience}"
    Score these four dimensions (0-100), then derive overall_score as their average:
    AUDIENCE CLARITY (principle: "audience"): Does the content serve the intended audience? Right level, language, needs?
    NAMING & LANGUAGE (principle: "naming"): Words the intended audience would use mid-problem, or internal jargon?
    RELEVANCE & NEED (principle: "relevance"): Real user need for this audience, timely and accurate, or internally motivated?
    INFORMATION TYPE FIT (principle: "info_type"): Clearly fits one of:
    1. What's new 2. Getting started 3. Learning 4. Doing 5. Fixing
    Respond ONLY with a valid JSON object, no markdown, no preamble:
    {"title":"page title","overall_score":72,"dimensions":[{"name":"Audience clarity","principle":"audience","score":80,"verdict":"one sentence referencing intended audience","suggestion":"one concrete improvement"},{"name":"Naming & language","principle":"naming","score":65,"verdict":"...","suggestion":"..."},{"name":"Relevance & need","principle":"relevance","score":70,"verdict":"...","suggestion":"..."},{"name":"Information type fit","principle":"info_type","score":75,"verdict":"...","suggestion":"...","detected_type":"Learning"}],"recommendation":"2-3 sentences referencing the intended audience specifically"}`;
    function setStatus(msg, isError) {
    const el = document.getElementById('statusMsg');
    el.style.display = 'block';
    el.className = 'status' + (isError ? ' error' : '');
    el.innerHTML = msg;
    }
    async function runScore() {
    const apiKey = document.getElementById('apiKey').value.trim();
    const audience = document.getElementById('audienceInput').value.trim();
    const url = document.getElementById('pageUrl').value.trim();
    const pasted = document.getElementById('pageContent').value.trim();
    let valid = true;
    ['apiKey','audienceInput'].forEach(id => {
    const val = document.getElementById(id).value.trim();
    document.getElementById(id).classList.toggle('error', !val);
    if (!val) valid = false;
    });
    if (activeTab === 'url' && !url) { document.getElementById('pageUrl').classList.add('error'); valid = false; }
    if (activeTab === 'paste' && !pasted) { document.getElementById('pageContent').classList.add('error'); valid = false; }
    if (!valid) { setStatus('Please fill in all required fields.', true); return; }
    document.getElementById('scoreBtn').disabled = true;
    document.getElementById('results').style.display = 'none';
    // Build request — URL mode uses web_fetch tool, paste mode sends content directly
    let requestBody;
    if (activeTab === 'url') {
    setStatus('<span class="spinner"></span> Fetching page and analysing…');
    requestBody = {
    model: 'claude-sonnet-4-20250514',
    max_tokens: 1400,
    system: SYSTEM(audience),
    tools: [{ type: 'web_fetch_20250910', name: 'web_fetch', max_uses: 3 }],
    messages: [{ role: 'user', content: `Please fetch and score this page: ${url}` }]
    };
    } else {
    setStatus('<span class="spinner"></span> Analysing against editorial principles…');
    requestBody = {
    model: 'claude-sonnet-4-20250514',
    max_tokens: 1400,
    system: SYSTEM(audience),
    messages: [{ role: 'user', content: `Page content:\n\n${pasted.slice(0, 14000)}` }]
    };
    }
    try {
    // For URL mode we may need to handle tool_use turns
    let messages = requestBody.messages;
    let analysis = null;
    for (let turn = 0; turn < 5; turn++) {
    const resp = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
    'Content-Type': 'application/json',
    'x-api-key': apiKey,
    'anthropic-version': '2023-06-01',
    'anthropic-beta': 'web-fetch-2025-09-10',
    'anthropic-dangerous-direct-browser-calls': 'true'
    },
    body: JSON.stringify({ ...requestBody, messages })
    });
    const data = await resp.json();
    if (data.error) throw new Error(data.error.message);
    // Collect text blocks
    const textBlocks = (data.content || []).filter(b => b.type === 'text');
    const toolUseBlocks = (data.content || []).filter(b => b.type === 'tool_use');
    if (data.stop_reason === 'end_turn' || toolUseBlocks.length === 0) {
    // Parse JSON from the final text response
    const text = textBlocks.map(b => b.text).join('');
    const clean = text.replace(/```json|```/g, '').trim();
    const s = clean.indexOf('{'), e = clean.lastIndexOf('}');
    if (s === -1) throw new Error('No JSON found in response');
    analysis = JSON.parse(clean.slice(s, e + 1));
    break;
    }
    // Tool use turn — append assistant message and tool results, then continue
    messages = [
    ...messages,
    { role: 'assistant', content: data.content },
    {
    role: 'user',
    content: toolUseBlocks.map(tb => ({
    type: 'tool_result',
    tool_use_id: tb.id,
    content: 'Tool executed — content has been fetched.'
    }))
    }
    ];
    }
    if (!analysis) throw new Error('Could not parse a score from the response.');
    renderResults(analysis, audience, url || '');
    document.getElementById('statusMsg').style.display = 'none';
    } catch(err) {
    setStatus('Error: ' + (err.message || 'Something went wrong. Check your API key and try again.'), true);
    }
    document.getElementById('scoreBtn').disabled = false;
    }
    const ICONS = {
    audience: `<svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M17 21v-2a4 4 0 0 0-4-4H5a4 4 0 0 0-4 4v2"/><circle cx="9" cy="7" r="4"/><path d="M23 21v-2a4 4 0 0 0-3-3.87"/><path d="M16 3.13a4 4 0 0 1 0 7.75"/></svg>`,
    naming: `<svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M20.59 13.41l-7.17 7.17a2 2 0 0 1-2.83 0L2 12V2h10l8.59 8.59a2 2 0 0 1 0 2.82z"/><line x1="7" y1="7" x2="7.01" y2="7"/></svg>`,
    relevance:`<svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><rect x="3" y="4" width="18" height="18" rx="2" ry="2"/><line x1="16" y1="2" x2="16" y2="6"/><line x1="8" y1="2" x2="8" y2="6"/><line x1="3" y1="10" x2="21" y2="10"/><path d="M9 16l2 2 4-4"/></svg>`,
    info_type:`<svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><line x1="8" y1="6" x2="21" y2="6"/><line x1="8" y1="12" x2="21" y2="12"/><line x1="8" y1="18" x2="21" y2="18"/><line x1="3" y1="6" x2="3.01" y2="6"/><line x1="3" y1="12" x2="3.01" y2="12"/><line x1="3" y1="18" x2="3.01" y2="18"/></svg>`
    };
    function sc(s) { return s >= 75 ? 'good' : s >= 50 ? 'warn' : 'bad'; }
    function sl(s) { return s >= 75 ? 'Good' : s >= 50 ? 'Needs work' : 'Poor'; }
    function renderResults(a, audience, url) {
    const cat = sc(a.overall_score);
    document.getElementById('pageTitle').textContent = a.title || url || 'Untitled page';
    document.getElementById('overallBadge').innerHTML = `<div class="score-num color-${cat}">${a.overall_score}</div><div class="score-label color-${cat}">${sl(a.overall_score)}</div>`;
    document.getElementById('audienceChip').innerHTML = `<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M20 21v-2a4 4 0 0 0-4-4H8a4 4 0 0 0-4 4v2"/><circle cx="12" cy="7" r="4"/></svg><span><strong>Scored for:</strong> ${audience}</span>`;
    document.getElementById('scoreGrid').innerHTML = a.dimensions.map(d => {
    const c = sc(d.score);
    return `<div class="score-card"><div class="dim-label">${d.name}</div><div class="dim-num color-${c}">${d.score}</div><div class="dim-verdict color-${c}">${sl(d.score)}</div></div>`;
    }).join('');
    document.getElementById('dimCards').innerHTML = a.dimensions.map(d => {
    const c = sc(d.score);
    const icon = ICONS[d.principle] || ICONS.info_type;
    return `<div class="dim-detail">
    <div class="dim-detail-header">
    <div class="dim-icon icon-${c}">${icon}</div>
    <h3>${d.name}${d.detected_type ? ` <small>(${d.detected_type})</small>` : ''}</h3>
    <span class="dim-badge bg-${c} color-${c}">${d.score}/100</span>
    </div>
    <div class="bar-track"><div class="bar-fill bar-${c}" style="width:${Math.max(3,d.score)}%"></div></div>
    <p class="dim-verdict-text">${d.verdict}</p>
    <div class="dim-suggestion"><svg width="13" height="13" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="12" cy="12" r="10"/><line x1="12" y1="8" x2="12" y2="12"/><line x1="12" y1="16" x2="12.01" y2="16"/></svg><span>${d.suggestion}</span></div>
    </div>`;
    }).join('');
    document.getElementById('recText').textContent = a.recommendation;
    document.getElementById('inputSection').style.display = 'none';
    document.getElementById('results').style.display = 'block';
    window.scrollTo({ top: 0, behavior: 'smooth' });
    }
